Recall the following useful facts about $D_{n}$ (which you discovered in a previous problem set).
\begin{itemize}
    \item Suppose $n$ is even. The conjugacy classes of $D_{n}$ are:
    \begin{align*}
        \{e\},  \{r,r^{n-1}\},  \{r^{2},r^{n-2}\},  \cdots ,  \{r^{\frac{n}{2}-1},r^{\frac{n}{2}+1}\}, \{r^{\frac{n}{2}}\}, \{s,r^{2}s,\cdots,r^{n-2}s\}, \{rs,r^{3}s,\cdots,r^{n-1}s\}.
    \end{align*}
    \item Suppose $n$ is odd. The conjugacy classes of $D_{n}$ are:
    \begin{align*}
        \{e\},  \{r,r^{n-1}\},  \{r^{2},r^{n-2}\},  \cdots , \{r^{\frac{n-1}{2}},r^{\frac{n+1}{2}}\},  \{s,rs,r^{2}s,\cdots,r^{n-1}s\}.
    \end{align*}
\end{itemize}
