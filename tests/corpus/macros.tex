\newcommand{\inner}[2]{\langle #1, #2 \rangle}

\newcommand{\diff}[2]{\frac{d #1}{d #2}}

\begin{document}
\section{Custom macros}
For all $\eps > 0$ and $x \in \R^n$, the norm $\norm{x}$ and inner product
$\inner{x}{y}$ are defined. The derivative is
\[
\diff{f}{x} = \lim_{h \to 0} \frac{f(x+h) - f(x)}{h}.
\]
